# The PDF backends the CORPUS INTAKE path needs, and nothing else needs.
#
#     python3 -m pip install -r requirements-lock.txt -r requirements-intake.txt
#     sudo apt-get install -y poppler-utils
#
# Kept apart from `requirements-lock.txt` on purpose. The readers, the grid
# engine and the finalizer never open a PDF - they are given rasters - so a
# machine that only digitizes figures does not need any of this, and a lock
# file that demanded it would make the core install larger than the core work.
#
# THE SPLIT IS WHY CI HAS TWO JOBS. `tests/test_corpus_intake.py` SKIPs its
# adapter, per-status, renderer and crop sections where no backend is installed,
# so the `core` job (backends removed) and the `full` job (this file installed)
# run different numbers of scenarios on purpose; `full` also checks that the PDF
# sections actually ran.
#
# poppler-utils is not a Python package and cannot be pinned here. The
# `full` CI job records `pdftotext -v` in its log, so a runner image that
# changes it is visible rather than mysterious - and the PDF sections are gated on
# poppler being PRESENT, which the job installs explicitly rather than
# inheriting from whatever `ubuntu-latest` happens to ship.
pdfminer.six==20251230
pypdf==3.17.4
# TESSERACT IS THE SAME KIND OF BACKEND AS POPPLER, and `test_tick_ocr` has
# three scenarios that read real glyphs. Without it they SKIP - three scenarios
# that ran nowhere but a developer's machine, which is what the two-profile
# job split exists to make visible. The apt package `tesseract-ocr` is the other
# half and cannot be pinned here; the `full` job installs it explicitly and
# records `tesseract --version` in its log, for the same reason it records
# `pdftotext -v`.
pytesseract==0.3.13
